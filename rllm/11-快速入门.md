# 快速入门

## 安装

```bash
pip install rllm
```

## 5 分钟上手

### 1. 注册数据集

```python
from rllm.data import DatasetRegistry
from datasets import load_dataset

data = load_dataset("gsm8k", "main", split="train")
DatasetRegistry.register_dataset(name="gsm8k", data=data, split="train")
```

### 2. 创建 Agent

```python
from rllm.agents import Agent

agent = Agent(
    name="MathSolver",
    model="qwen/qwen2.5-7b",
    system_prompt="你是一个数学解题助手。",
)
```

### 3. 创建 Environment

```python
from rllm.environments import Environment

env = Environment(
    name="MathEnv",
    datasets=["gsm8k"],
    reward_fn=math_reward,
)
```

### 4. 创建 Workflow

```python
from rllm.workflows import GRPOWorkflow

workflow = GRPOWorkflow(
    agent=agent,
    env=env,
    num_rollouts=4,
)
```

### 5. 训练

```python
from rllm.trainer import GRPOTrainer

trainer = GRPOTrainer(
    model=agent.model,
    workflow=workflow,
    learning_rate=1e-5,
)

trainer.train(num_steps=1000)
```

## 完整示例

```python
from rllm import Agent, Environment, GRPOWorkflow, GRPOTrainer
from rllm.data import DatasetRegistry

# 1. 准备数据
DatasetRegistry.register_dataset(name="gsm8k", data=load_dataset("gsm8k", split="train"), split="train")

# 2. 创建组件
agent = Agent(name="Solver", model="qwen/qwen2.5-7b")
env = Environment(name="MathEnv", datasets=["gsm8k"], reward_fn=math_reward)
workflow = GRPOWorkflow(agent=agent, env=env, num_rollouts=4)

# 3. 训练
trainer = GRPOTrainer(model=agent.model, workflow=workflow)
trainer.train(num_steps=1000)
```
